function doGet(e) { return handleRequest(e); } function doPost(e) { return handleRequest(e); } function handleRequest(e) { try { var action = (e.parameter && e.parameter.action) || 'sync'; var yr = (e.parameter && e.parameter.year) || new Date().getFullYear().toString(); var callback = (e.parameter && e.parameter.callback) || ''; var ss = SpreadsheetApp.getActiveSpreadsheet(); var result; if (action === 'ping') { result = {status:'ok', msg:'connected'}; } else if (action === 'fetch') { var data = loadSheets(ss, yr); result = {status:'ok', data: data}; } else if (action === 'sync') { var raw = (e.parameter && e.parameter.data) || (e.postData && e.postData.contents) || '{}'; var data = JSON.parse(raw.indexOf('%') !== -1 ? decodeURIComponent(raw) : raw); saveSheets(ss, yr, data); result = {status:'ok', saved: (data.leaves||[]).length + (data.rests||[]).length}; } else if (action === 'archive') { var raw = (e.parameter && e.parameter.data) || '{}'; var data = JSON.parse(raw.indexOf('%') !== -1 ? decodeURIComponent(raw) : raw); saveArchiveSheet(ss, yr, data); result = {status:'ok', archived: yr}; } else { result = {status:'error', msg:'unknown action'}; } return respond(result, callback); } catch(err) { return respond({status:'error', msg: err.toString()}, ''); } } function loadSheets(ss, yr) { var data = { year: yr, leaves: [], rests: [], totalAnnual: 21, lastModified: 0 }; var lSheet = ss.getSheetByName('LeaveData_'+yr) || ss.getSheetByName('الإجازات_'+yr) || ss.getSheetByName('اجازات_'+yr); if (lSheet) { var vals = lSheet.getDataRange().getValues(); for (var i = 1; i < vals.length; i++) { if (vals[i][0]) { data.leaves.push({ id: parseInt(vals[i][0]) || 0, date: vals[i][1] || '', type: vals[i][2] || 'سنوي', days: String(vals[i][3] || '1'), restRef: vals[i][4] || null, shift: vals[i][5] || null }); } } } var rSheet = ss.getSheetByName('RestData_'+yr) || ss.getSheetByName('بدلات_'+yr) || ss.getSheetByName('راحة_'+yr); if (rSheet) { var rvals = rSheet.getDataRange().getValues(); for (var j = 1; j < rvals.length; j++) { if (rvals[j][0]) { data.rests.push({ id: parseInt(rvals[j][0]) || 0, date: rvals[j][1] || '', days: String(rvals[j][2] || '1'), validDays: parseInt(rvals[j][3]) || 30, note: rvals[j][4] || '' }); } } } var sSheet = ss.getSheetByName('Settings_'+yr) || ss.getSheetByName('اعدادات_'+yr); if (sSheet) { var svals = sSheet.getDataRange().getValues(); for (var k = 0; k < svals.length; k++) { if (svals[k][0] && svals[k][0].indexOf && svals[k][0].indexOf('رصيد') !== -1) data.totalAnnual = parseFloat(svals[k][1]) || 21; if (svals[k][0] === 'lastModified') data.lastModified = parseInt(svals[k][1]) || 0; } } return data; } function saveSheets(ss, yr, data) { var lSheet = ss.getSheetByName('LeaveData_'+yr) || ss.getSheetByName('الإجازات_'+yr) || ss.getSheetByName('اجازات_'+yr) || ss.insertSheet('LeaveData_'+yr); lSheet.clearContents(); lSheet.getRange(1,1,1,7).setValues([['ID','Date','Type','Days','RestRef','Shift','Timestamp']]); (data.leaves || []).forEach(function(l) { lSheet.appendRow([l.id || Date.now(), l.date || '', l.type || 'سنوي', l.days || '1', l.restRef || '', l.shift || '', new Date(l.id || Date.now()).toLocaleString('ar-EG')]); }); var rSheet = ss.getSheetByName('RestData_'+yr) || ss.getSheetByName('بدلات_'+yr) || ss.getSheetByName('راحة_'+yr) || ss.insertSheet('RestData_'+yr); rSheet.clearContents(); rSheet.getRange(1,1,1,6).setValues([['ID','Date','Days','ValidDays','Note','Timestamp']]); (data.rests || []).forEach(function(r) { rSheet.appendRow([r.id || Date.now(), r.date || '', r.days || '1', r.validDays || 30, r.note || '', new Date(r.id || Date.now()).toLocaleString('ar-EG')]); }); var sSheet = ss.getSheetByName('Settings_'+yr) || ss.getSheetByName('اعدادات_'+yr) || ss.insertSheet('Settings_'+yr); sSheet.clearContents(); sSheet.getRange(1,1,4,2).setValues([ ['TotalAnnual', data.totalAnnual || 21], ['Year', yr], ['LastSync', new Date().toLocaleString('ar-EG')], ['lastModified', data.lastModified || Date.now()] ]); } function saveArchiveSheet(ss, yr, data) { // شيت مستقل باسم Archive_YYYY يحفظ البيانات القديمة بشكل دائم var sheetName = 'Archive_' + yr; var sh = ss.getSheetByName(sheetName) || ss.insertSheet(sheetName); sh.clearContents(); // Header sh.getRange(1,1,1,2).setValues([['السنة', yr]]); sh.getRange(2,1,1,2).setValues([['الرصيد السنوي', data.totalAnnual || 21]]); sh.getRange(3,1,1,2).setValues([['تاريخ الأرشفة', new Date().toLocaleString('ar-EG')]]); sh.getRange(4,1,1,2).setValues([['lastModified', data.lastModified || 0]]); // Leaves sh.getRange(6,1,1,7).setValues([['--- الإجازات ---','','','','','','']]); sh.getRange(7,1,1,7).setValues([['ID','Date','Type','Days','RestRef','Shift','Timestamp']]); var row = 8; (data.leaves || []).forEach(function(l) { sh.getRange(row++,1,1,7).setValues([[ l.id || '', l.date || '', l.type || 'سنوي', l.days || '1', l.restRef || '', l.shift || '', new Date(l.id || Date.now()).toLocaleString('ar-EG') ]]); }); // Rests row++; sh.getRange(row,1,1,5).setValues([['--- بدلات الراحة ---','','','','']]); row++; sh.getRange(row++,1,1,5).setValues([['ID','Date','Days','ValidDays','Note']]); (data.rests || []).forEach(function(r) { sh.getRange(row++,1,1,5).setValues([[ r.id || '', r.date || '', r.days || '1', r.validDays || 30, r.note || '' ]]); }); } function respond(obj, callback) { var json = JSON.stringify(obj); // لو في callback → JSONP (بيحل CORS تلقائياً) if (callback) { return ContentService.createTextOutput(callback + '(' + json + ')') .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService.createTextOutput(json) .setMimeType(ContentService.MimeType.JSON); }
